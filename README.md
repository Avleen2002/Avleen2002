```c
void pancakeSort(int* arr, int n) {
    int maxdex;
    while (n > 1) {
        maxdex = maxIndex(arr, n);
        if (maxdex != n) {
            flip(arr, maxdex);
            flip(arr, n-1);
        }
        n--;
    }
}
```
